## 二、部署模块详细设计

### 2.1 setup.sh 脚本设计

#### 2.1.1 功能描述

初始化Docker容器环境，安装基础软件包，配置SSH免密登录。

#### 2.1.2 输入输出

| 类型 | 内容 |
|-----|------|
| 输入 | 无（从环境变量读取容器列表） |
| 输出 | 初始化完成的容器环境 |
| 返回值 | 0=成功，非0=失败 |

#### 2.1.3 算法流程

```
算法: setup()
输入: 无
输出: 初始化结果

1. INSTANCES ← ["bigtop_hostname0", "bigtop_hostname1", "bigtop_hostname2", "bigtop_hostname3"]
2. FOR each i IN INSTANCES DO
3.   IF container_not_running(i) THEN
4.     print_error("Container not found: " + i)
5.     RETURN 1
6.   END IF
7. END FOR
8. 
9. FOR each i IN INSTANCES DO
10.   install_packages(i, ["sudo", "openssh-server", "openssh-clients", ...])
11.   enable_powertools(i)
12.   update_system(i)
13. END FOR
14.
15. FOR each i IN INSTANCES DO
16.   generate_ssh_key(i)
17.   start_sshd(i)
18. END FOR
19.
20. copy_public_key("bigtop_hostname0", "bigtop_hostname0.pub")
21. FOR each i IN INSTANCES DO
22.   copy_file("bigtop_hostname0.pub", i + ":/root/.ssh/authorized_keys")
23.   set_permissions(i, "/root/.ssh/authorized_keys", 600)
24. END FOR
25.
26. FOR each i IN INSTANCES DO
27.   FOR each j IN INSTANCES DO
28.     add_known_hosts(i, j)
29.   END FOR
30. END FOR
31.
32. RETURN 0
```

#### 2.1.4 核心代码

```bash
#!/bin/bash
set -ex

INSTANCES="bigtop_hostname0 bigtop_hostname1 bigtop_hostname2 bigtop_hostname3"

# 检查容器状态
for i in $INSTANCES ; do
  if [ "$(docker compose ps | grep -c $i)" == "0" ] ; then
    echo "ERROR: can't find docker container: $i"
    exit 1
  fi
done

# 安装基础软件包
for i in $INSTANCES ; do
  docker compose exec -it $i dnf install -y \
    sudo openssh-server openssh-clients which iproute net-tools \
    less vim-enhanced initscripts wget curl tar unzip git dnf-plugins-core
  docker compose exec -it $i dnf config-manager --set-enabled powertools 2>/dev/null || true
  docker compose exec -it $i dnf update -y || true
done

# 配置SSH
for i in $INSTANCES ; do
  docker compose exec -it $i bash -c \
    '[ -f /root/.ssh/id_rsa ] || ssh-keygen -t rsa -N "" -f /root/.ssh/id_rsa'
  docker compose exec -it $i systemctl enable sshd
  docker compose exec -it $i systemctl start sshd
done

# 分发公钥
docker compose cp bigtop_hostname0:/root/.ssh/id_rsa.pub bigtop_hostname0.pub
for i in $INSTANCES ; do
  docker compose cp bigtop_hostname0.pub $i:/root/.ssh/authorized_keys
  docker compose exec -it $i chmod 600 /root/.ssh/authorized_keys
done
rm -f bigtop_hostname0.pub
```

#### 2.1.5 异常处理

| 异常情况 | 处理方式 |
|---------|---------|
| 容器未运行 | 输出错误信息，退出脚本 |
| 软件包安装失败 | 记录警告，继续执行 |
| SSH配置失败 | 输出错误信息，退出脚本 |

#### 3.1.4 核心代码

```bash
#!/bin/bash
set -ex

REPO_DIR=/var/repo/ambari

echo "==> Setting up local YUM repository..."

# 创建仓库目录
mkdir -p ${REPO_DIR}

# 复制RPM包
echo "==> Copying RPMs..."
cp -r /var/repo/ambarirpm/* ${REPO_DIR}/ 2>/dev/null || true
cp -r /var/repo/bigtoprpm/*/* ${REPO_DIR}/ 2>/dev/null || true

# 生成仓库元数据

server {
    listen 80;
    server_name _;
    root /var/repo/ambari;
    location / {
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
    }
}
EOF

# 启动Nginx
systemctl enable nginx
systemctl start nginx

# 验证
curl -s http://localhost/repodata/repomd.xml > /dev/null && \
  echo "Repository is accessible" || \
  echo "WARNING: Repository not accessible"
```

```bash
#!/bin/bash
set -ex

STACKS_DIR=/var/lib/ambari-server/resources/stacks
STACK_HOME=${STACKS_DIR}/BIGTOP/3.5.0

# 查找父Stack
PARENT_STACK=$(ls "${STACKS_DIR}/BIGTOP/" | grep -v '3.5.0' | sort -V | tail -1)
echo "==> Using parent stack: BIGTOP/${PARENT_STACK}"

# 复制Stack
cp -r "${STACKS_DIR}/BIGTOP/${PARENT_STACK}" "${STACK_HOME}"

# 创建metainfo.xml
cat > "${STACK_HOME}/metainfo.xml" << EOF
<?xml version="1.0"?>
<metainfo>
  <versions>
    <active>true</active>
  </versions>
  <extends>${PARENT_STACK}</extends>
</metainfo>
EOF

# 创建repoinfo.xml
mkdir -p "${STACK_HOME}/repos"
cat > "${STACK_HOME}/repos/repoinfo.xml" << 'EOF'
<?xml version="1.0"?>
<reposinfo>
  <latest>http://bigtop_hostname0/repodata/repomd.xml</latest>
  <os family="redhat8">
    <repo>
      <baseurl>http://bigtop_hostname0</baseurl>
      <repoid>BIGTOP-3.5.0</repoid>
      <reponame>Bigtop 3.5.0</reponame>
      <unique>true</unique>
    </repo>
  </os>
</reposinfo>
EOF

# 更新版本号
update_service_version() {
  local service=$1
  local version=$2
  local metainfo="${STACK_HOME}/services/${service}/metainfo.xml"
  if [ -f "$metainfo" ]; then
    sed -i "s|<version>.*</version>|<version>${version}</version>|g" "$metainfo"
    echo "    Updated ${service} to ${version}"
  fi
}

update_service_version HDFS 3.3.6
update_service_version YARN 3.3.6
update_service_version HIVE 4.0.1
update_service_version HBASE 2.6.2
update_service_version SPARK2 3.5.6
```

```bash
#!/bin/bash
```

```bash
if [ -f /etc/hive/conf/hive-site.xml ]; then
  sed -i 's|<value> </value>|<value></value>|g' /etc/hive/conf/hive-site.xml

  sed -i 's/return " "/return ""/g' "$SERVICE_ADVISOR"
  echo "    Fixed service_advisor.py"

# 修复3: hive_check.py
echo "[3/7] Fixing hive_check.py..."
HIVE_CHECK="/usr/lib/ambari-agent/lib/resource_management/libraries/functions/hive_check.py"

  sed -i 's/beeline_url.append("auth=noSasl")/beeline_url.append("auth=none")/g' "$HIVE_CHECK"
  echo "    Fixed hive_check.py"

# 修复4: HDFS权限
echo "[4/7] Fixing HDFS permissions..."
hdfs dfs -mkdir -p /warehouse/tablespace/managed/hive 2>/dev/null || true
hdfs dfs -mkdir -p /warehouse/tablespace/external/hive 2>/dev/null || true
hdfs dfs -chown -R hive:hadoop /warehouse/tablespace 2>/dev/null || true

  if [ -n "$SCHEMA_FILE" ]; then

    echo "    Schema initialized"
  fi
fi

# 修复6: 标记文件
echo "[6/7] Creating marker file..."
mkdir -p /etc/hive
touch /etc/hive/sys.db.created

echo "==> Fix completed"
```

#### 7.1.4 核心代码

```python
#!/usr/bin/env python3
import mysql.connector
import sys
import os

VDF_PATH = '/var/repo/ambari/bigtop-3.5.0-vdf.xml'

def main():
    if not os.path.exists(VDF_PATH):
        print(f"ERROR: VDF file not found: {VDF_PATH}")
        sys.exit(1)
    
    with open(VDF_PATH, 'r') as f:
        vdf_xml = f.read()
    
    try:
        conn = mysql.connector.connect(
            host='localhost',
            user='ambari',
            password='ambari',
            database='ambari'
        )
        cursor = conn.cursor()
        
        # 检查是否已存在
        cursor.execute("SELECT COUNT(*) FROM repo_version WHERE version='3.5.0.0-1'")
        if cursor.fetchone()[0] > 0:
            print("Version already exists")
```

### 7.2 fix_secrets.py 设计

#### 7.2.1 功能描述

修复配置文件中的敏感信息和空格值问题。

#### 7.2.2 核心代码

```python
#!/usr/bin/env python3
import xml.etree.ElementTree as ET
import os

def fix_hive_site():
    """修复hive-site.xml中的配置问题"""
    hive_site = '/etc/hive/conf/hive-site.xml'
    
    if not os.path.exists(hive_site):
        print("hive-site.xml not found")
        return
    
    tree = ET.parse(hive_site)
    root = tree.getroot()
    modified = False
    
    for prop in root.findall('property'):
        name = prop.find('name')
        value = prop.find('value')
        
        if name is not None and value is not None:
            # 修复空格值
            if value.text and value.text.strip() == '' and len(value.text) > 0:
                value.text = ''
                modified = True
                print(f"Fixed: {name.text}")
    
    if modified:
        tree.write(hive_site, encoding='utf-8', xml_declaration=True)
        print("hive-site.xml fixed")

if __name__ == '__main__':
    fix_hive_site()
```
