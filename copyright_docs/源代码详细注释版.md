### 1.1 setup.sh - 节点环境初始化脚本

```bash
#!/bin/bash
#===============================================================================
# 文件名称: setup.sh
# 功能描述: HodoopForge节点环境初始化脚本
#           用于初始化Docker容器环境，安装基础软件包，配置SSH免密登录
# 
# 作者: HodoopForge开发团队
# 创建日期: 2026-04
# 版本: V1.0
#
# 使用方法:
#   ./setup.sh
#
# 前置条件:
#   1. Docker环境已安装
#   2. docker-compose.yml已配置
#   3. 容器已启动 (docker compose up -d)
#
# 执行结果:
#   1. 所有容器安装基础软件包
#   2. SSH服务启动
#   3. SSH免密登录配置完成
#===============================================================================

# 启用脚本调试模式
# -e: 当命令返回非零值时立即退出
# -x: 打印执行的每条命令
set -ex

#===============================================================================
# 全局变量定义
#===============================================================================

# 定义所有容器实例名称
# 这些名称与docker-compose.yml中定义的服务名称一致
INSTANCES="bigtop_hostname0 bigtop_hostname1 bigtop_hostname2 bigtop_hostname3"

#===============================================================================
# 函数定义
#===============================================================================

# 函数: check_containers
# 功能: 检查所有容器是否正常运行
# 参数: 无
# 返回: 0表示成功，1表示失败
check_containers() {
    echo "==> [步骤1] 检查容器运行状态..."
    
    for i in $INSTANCES ; do
        # 使用docker compose ps检查容器状态
        # grep -c 统计匹配行数
        if [ "$(docker compose ps | grep -c $i)" == "0" ] ; then
            echo "错误: 找不到容器 $i"
            echo "请先执行: docker compose up -d"
            return 1
        fi
        echo "    容器 $i 状态正常"
    done
    
    echo "==> 所有容器运行正常"
    return 0
}

# 函数: install_packages
# 功能: 在所有容器中安装基础软件包
# 参数: 无
# 返回: 无
install_packages() {
    echo "==> [步骤2] 安装基础软件包..."
    
    for i in $INSTANCES ; do
        echo "    正在为容器 $i 安装软件包..."
        
        # 安装系统管理工具
        # sudo: 权限管理
        # openssh-server/openssh-clients: SSH服务
        # which, iproute, net-tools: 网络工具
        # less, vim-enhanced: 文本编辑器
        docker compose exec -it $i dnf install -y \
            sudo openssh-server openssh-clients which iproute net-tools \
            less vim-enhanced
        
        # 安装下载和压缩工具
        # initscripts: 系统初始化脚本
        # wget, curl: 下载工具
        # tar, unzip: 压缩解压工具
        # git: 版本控制
        docker compose exec -it $i dnf install -y \
            initscripts wget curl tar unzip git
        
        # 安装DNF插件
        # dnf-plugins-core: DNF核心插件，支持config-manager等命令
        docker compose exec -it $i dnf install -y dnf-plugins-core
        
        # 启用Powertools仓库
        # Powertools包含额外的开发包和工具
        # 不同发行版名称可能不同: powertools 或 rocky-powertools
        docker compose exec -it $i dnf config-manager --set-enabled powertools 2>/dev/null \
            || docker compose exec -it $i dnf config-manager --set-enabled rocky-powertools 2>/dev/null \
            || echo "    Powertools已启用或不需要"
        
        # 更新系统软件包
        # 某些包可能更新失败，使用 || true 继续执行
        docker compose exec -it $i dnf update -y || echo "    部分软件包更新失败，继续执行"
    done
    
    echo "==> 基础软件包安装完成"
}

# 函数: configure_ssh
# 功能: 配置SSH服务和免密登录
# 参数: 无
# 返回: 无
configure_ssh() {
    echo "==> [步骤3] 配置SSH服务..."
    
    # 为每个容器生成SSH密钥对
    for i in $INSTANCES ; do
        echo "    为容器 $i 配置SSH..."
        
        # 检查并生成SSH密钥
        # 如果密钥不存在，则生成新的RSA密钥对
        # -N "": 不设置密码短语
        # -f: 指定密钥文件路径
        docker compose exec -it $i bash -c \
            '[ -f /root/.ssh/id_rsa ] || ssh-keygen -t rsa -N "" -f /root/.ssh/id_rsa'
        
        # 启用SSH服务开机自启
        docker compose exec -it $i systemctl enable sshd
        
        # 启动SSH服务
        docker compose exec -it $i systemctl start sshd
    done
    
    echo "==> SSH服务配置完成"
}

# 函数: distribute_keys
# 功能: 分发SSH公钥到所有节点
# 参数: 无
# 返回: 无
distribute_keys() {
    echo "==> [步骤4] 分发SSH公钥..."
    
    # 从hostname0容器复制公钥到宿主机
    # 这将作为所有节点的授权密钥
    docker compose cp bigtop_hostname0:/root/.ssh/id_rsa.pub bigtop_hostname0.pub
    
    # 将公钥复制到所有节点的authorized_keys文件
    for i in $INSTANCES ; do
        echo "    将公钥复制到容器 $i..."
        
        # 复制公钥文件
        docker compose cp bigtop_hostname0.pub $i:/root/.ssh/authorized_keys
        
        # 设置正确的文件权限
        # chown: 设置文件所有者
        # chmod 600: 只有所有者可读写
        docker compose exec -it $i chown root:root /root/.ssh/authorized_keys
        docker compose exec -it $i chmod 600 /root/.ssh/authorized_keys
    done
    
    # 清理临时文件
    rm -f bigtop_hostname0.pub
    
    echo "==> SSH公钥分发完成"
}

# 函数: configure_known_hosts
# 功能: 配置SSH known_hosts以避免首次连接确认
# 参数: 无
# 返回: 无
configure_known_hosts() {
    echo "==> [步骤5] 配置known_hosts..."
    
    # 为每个容器配置所有其他容器的主机密钥
    for i in $INSTANCES ; do
        for j in $INSTANCES ; do
            # 使用ssh-keyscan获取主机密钥
            # -H: 哈希主机名，提高安全性
            # 2>/dev/null: 忽略错误输出
            docker compose exec -it $i bash -c \
                "ssh-keyscan -H $j.bigtop.apache.org >> /root/.ssh/known_hosts 2>/dev/null || true"
        done
    done
    
    echo "==> known_hosts配置完成"
}

#===============================================================================
# 主程序入口
#===============================================================================

echo "=========================================="
echo "  HodoopForge 节点环境初始化"
echo "  版本: V1.0"
echo "  日期: 2026-04"
echo "=========================================="
echo ""

# 执行初始化步骤
check_containers || exit 1
install_packages
configure_ssh
distribute_keys
configure_known_hosts

echo ""
echo "=========================================="
echo "  节点环境初始化完成!"
echo "=========================================="
echo ""
echo "下一步操作:"
echo "  1. 执行 ./setup-ambari-repo.sh 构建本地仓库"
echo "  2. 执行 ./install-ambari-server.sh 安装Ambari Server"
echo ""
```

### 1.2 setup-ambari-repo-script.sh - 本地仓库构建脚本

```bash
#!/bin/bash
#===============================================================================
# 文件名称: setup-ambari-repo-script.sh
# 功能描述: HodoopForge本地YUM仓库构建脚本
#           用于构建本地YUM仓库，提供离线安装支持
# 
# 作者: HodoopForge开发团队
# 创建日期: 2026-04
# 版本: V1.0
#
# 使用方法:
#   ./setup-ambari-repo-script.sh
#
# 前置条件:
#   1. RPM包已放置在指定目录
#   2. Nginx已安装或可通过DNF安装
#
# 执行结果:
#   1. YUM仓库构建完成
#   2. Nginx HTTP服务启动
#   3. 仓库可通过HTTP访问
#===============================================================================

set -ex

#===============================================================================
# 全局变量定义
#===============================================================================

# 仓库根目录
# 所有RPM包将复制到此目录
REPO_DIR=/var/repo/ambari

# Ambari RPM包源目录
# 这些包来自ambarirpm目录
AMBARI_RPM_SRC=/var/repo/ambarirpm

# Bigtop RPM包源目录
# 这些包来自bigtoprpm目录
BIGTOP_RPM_SRC=/var/repo/bigtoprpm

# Nginx配置文件路径
NGINX_CONF=/etc/nginx/conf.d/repo.conf

#===============================================================================
# 主程序
#===============================================================================

echo "=========================================="
echo "  HodoopForge 本地仓库构建"
echo "=========================================="
echo ""

#-------------------------------------------------------------------------------
# 步骤1: 创建仓库目录
#-------------------------------------------------------------------------------
echo "==> [步骤1] 创建仓库目录..."

# 创建仓库根目录
# -p: 如果父目录不存在则创建
mkdir -p ${REPO_DIR}

# 创建子目录结构
mkdir -p ${REPO_DIR}/repodata

echo "    仓库目录创建完成: ${REPO_DIR}"

#-------------------------------------------------------------------------------
# 步骤2: 复制RPM包
#-------------------------------------------------------------------------------
echo "==> [步骤2] 复制RPM包..."

# 复制Ambari RPM包
# 这些包包括: ambari-server, ambari-agent, ambari-web等
if [ -d "${AMBARI_RPM_SRC}" ]; then
    echo "    复制Ambari RPM包..."
    cp -r ${AMBARI_RPM_SRC}/* ${REPO_DIR}/ 2>/dev/null || true
    AMBARI_COUNT=$(ls ${AMBARI_RPM_SRC}/*.rpm 2>/dev/null | wc -l)
    echo "    已复制 ${AMBARI_COUNT} 个Ambari RPM包"
fi

# 复制Bigtop RPM包
# Bigtop RPM包按组件分类存放，需要遍历子目录
if [ -d "${BIGTOP_RPM_SRC}" ]; then
    echo "    复制Bigtop RPM包..."
    
    # 遍历所有子目录
    for component_dir in ${BIGTOP_RPM_SRC}/*; do
        if [ -d "$component_dir" ]; then
            # 复制该组件目录下的所有RPM包
            cp -r $component_dir/*.rpm ${REPO_DIR}/ 2>/dev/null || true
        fi
    done
    
    BIGTOP_COUNT=$(find ${BIGTOP_RPM_SRC}
 -name "*.rpm" 2>/dev/null | wc -l)
    echo "    已复制 ${BIGTOP_COUNT} 个Bigtop RPM包"
fi

# 统计总RPM包数量
TOTAL_COUNT=$(ls ${REPO_DIR}/*.rpm 2>/dev/null | wc -l)
echo "    仓库中共有 ${TOTAL_COUNT} 个RPM包"

#-------------------------------------------------------------------------------
# 步骤3: 生成仓库元数据
#-------------------------------------------------------------------------------
echo "==> [步骤3] 生成仓库元数据..."

# 检查createrepo_c是否可用
# createrepo_c是createrepo的C语言实现，性能更好
if command -v createrepo_c &> /dev/null; then
    createrepo_c ${REPO_DIR}
    echo "    使用createrepo_c生成元数据"
else
    # 回退到createrepo
    createrepo ${REPO_DIR}
    echo "    使用createrepo生成元数据"
fi

# 验证元数据文件
if [ -f "${REPO_DIR}/repodata/repomd.xml" ]; then
    echo "    仓库元数据生成成功"
else
    echo "    错误: 仓库元数据生成失败"
    exit 1
fi

#-------------------------------------------------------------------------------
# 步骤4: 配置Nginx HTTP服务
#-------------------------------------------------------------------------------
echo "==> [步骤4] 配置Nginx HTTP服务..."

# 检查Nginx是否已安装
if ! command -v nginx &> /dev/null; then
    echo "    安装Nginx..."
    dnf install -y nginx
fi

# 创建Nginx虚拟主机配置
# 此配置将仓库目录作为HTTP服务提供
cat > ${NGINX_CONF} << 'EOF'
# HodoopForge 本地YUM仓库配置
# 提供HTTP访问本地RPM包的能力

server {
    # 监听80端口
    listen 80;
    
    # 接受所有主机名请求
    server_name _;
    
    # 仓库根目录
    root /var/repo/ambari;
    
    # 位置配置
    location / {
        # 启用目录浏览
        autoindex on;
        
        # 显示文件大小（KB, MB等）
        autoindex_exact_size off;
        
        # 显示本地时间
        autoindex_localtime on;
    }
}
EOF

echo "    Nginx配置文件已创建: ${NGINX_CONF}"

#-------------------------------------------------------------------------------
# 步骤5: 启动Nginx服务
#-------------------------------------------------------------------------------
echo "==> [步骤5] 启动Nginx服务..."

# 启用Nginx开机自启
systemctl enable nginx

# 启动Nginx服务
systemctl start nginx

# 检查Nginx状态
if systemctl is-active --quiet nginx; then
    echo "    Nginx服务启动成功"
else
    echo "    错误: Nginx服务启动失败"
    exit 1
fi

#-------------------------------------------------------------------------------
# 步骤6: 验证仓库可访问性
#-------------------------------------------------------------------------------
echo "==> [步骤6] 验证仓库可访问性..."

# 测试仓库URL
REPO_URL="http://localhost/repodata/repomd.xml"

if curl -s ${REPO_URL} > /dev/null; then
    echo "    仓库访问测试成功"
    echo "    仓库URL: http://bigtop_hostname0/"
else
    echo "    警告: 仓库访问测试失败，请检查Nginx配置"
fi

#-------------------------------------------------------------------------------
# 步骤7: 配置本地YUM源
#-------------------------------------------------------------------------------
echo "==> [步骤7] 配置本地YUM源..."

# 创建YUM源配置文件
cat > /etc/yum.repos.d/bigtop.repo << 'EOF'
[bigtop-3.5.0]
name=Bigtop 3.5.0
baseurl=http://bigtop_hostname0
enabled=1
gpgcheck=0
EOF

echo "    YUM源配置完成"

echo ""
echo "=========================================="
echo "  本地仓库构建完成!"
echo "=========================================="
echo ""
echo "仓库信息:"
echo "  URL: http://bigtop_hostname0/"
echo "  RPM包数量: ${TOTAL_COUNT}"
echo "  目录: ${REPO_DIR}"
echo ""
```

### 1.3 install-ambari-server-script.sh - Ambari Server安装脚本

```bash
#!/bin/bash
#===============================================================================
# 文件名称: install-ambari-server-script.sh
# 功能描述: HodoopForge Ambari Server安装脚本
#           用于安装和配置Ambari Server，包括MySQL数据库配置
# 
# 作者: HodoopForge开发团队
# 创建日期: 2026-04
# 版本: V1.0
#
# 使用方法:
#   ./install-ambari-server-script.sh
#
# 前置条件:
#   1. 本地YUM仓库已构建
#   2. 容器环境已初始化
#
# 执行结果:
#   1. MySQL数据库安装配置完成
#   2. Ambari Server安装完成
#   3. Bigtop 3.5.0 Stack注入完成
#   4. Ambari Server服务启动
#===============================================================================

set -ex

#===============================================================================
# 全局变量定义
#===============================================================================

# 数据库配置
DB_HOST="localhost"
DB_PORT="3306"
DB_NAME="ambari"
DB_USER="ambari"
DB_PASS="ambari"

# Hive数据库配置
HIVE_DB_NAME="hive"
HIVE_DB_USER="hive"
HIVE_DB_PASS="hive"

# Ambari配置
AMBARI_PORT="8080"
JAVA_HOME="/usr/lib/jvm/java-1.8.0-openjdk"

# Stack配置
STACKS_DIR=/var/lib/ambari-server/resources/stacks
STACK_HOME=${STACKS_DIR}/BIGTOP/3.5.0

#===============================================================================
# 函数定义
#===============================================================================

# 函数: install_mysql
# 功能: 安装和配置MySQL数据库
install_mysql() {
    echo "==> [步骤1] 安装MySQL数据库..."
    
    # 安装MySQL服务器
    dnf install -y mysql-server
    
    # 启用MySQL开机自启
    systemctl enable mysqld
    
    # 启动MySQL服务
    systemctl start mysqld
    
    # 等待MySQL服务就绪
    sleep 5
    
    echo "    MySQL安装完成"
}

# 函数: create_databases
# 功能: 创建Ambari和Hive数据库
create_databases() {
    echo "==> [步骤2] 创建数据库..."
    
    # 创建Ambari数据库
    # CHARACTER SET utf8mb4: 支持完整的Unicode字符集
    mysql -e "CREATE DATABASE ${DB_NAME} CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;"
    
    # 创建Ambari用户并授权
    mysql -e "CREATE USER '${DB_USER}'@'localhost' IDENTIFIED BY '${DB_PASS}';"
    mysql -e "GRANT ALL PRIVILEGES ON ${DB_NAME}.* TO '${DB_USER}'@'localhost';"
    
    # 创建Hive数据库
    mysql -e "CREATE DATABASE ${HIVE_DB_NAME} CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;"
    
    # 创建Hive用户并授权
    mysql -e "CREATE USER '${HIVE_DB_USER}'@'localhost' IDENTIFIED BY '${HIVE_DB_PASS}';"
    mysql -e "GRANT ALL PRIVILEGES ON ${HIVE_DB_NAME}.* TO '${HIVE_DB_USER}'@'localhost';"
    
    # 刷新权限
    mysql -e "FLUSH PRIVILEGES;"
    
    echo "    数据库创建完成"
    echo "    - Ambari数据库: ${DB_NAME}"
    echo "    - Hive数据库: ${HIVE_DB_NAME}"
}

# 函数: install_ambari_server
# 功能: 安装Ambari Server
install_ambari_server() {
    echo "==> [步骤3] 安装Ambari Server..."
    
    # 安装Ambari Server RPM包
    dnf install -y ambari-server
    
    echo "    Ambari Server安装完成"
}

# 函数: configure_ambari_server
# 功能: 配置Ambari Server
configure_ambari_server() {
    echo "==> [步骤4] 配置Ambari Server..."
    
    # 执行Ambari Server setup
    # --silent: 静默模式，不提示用户输入
    # --java-home: 指定Java安装路径
    # --database: 指定数据库类型
    ambari-server setup --silent \
        --java-home ${JAVA_HOME} \
        --database mysql \
        --databasehost ${DB_HOST} \
        --databaseport ${DB_PORT} \
        --databasename ${DB_NAME} \
        --databaseusername ${DB_USER} \
        --databasepassword ${DB_PASS}
    
    echo "    Ambari Server配置完成"
}

# 函数: import_schema
# 功能: 导入Ambari数据库schema
import_schema() {
    echo "==> [步骤5] 导入数据库schema..."
    
    # Ambari DDL SQL文件路径
    SCHEMA_FILE="/var/lib/ambari-server/resources/Ambari-DDL-MySQL-CREATE.sql"
    
    # 导入schema
    mysql -u${DB_USER} -p${DB_PASS} ${DB_NAME} < ${SCHEMA_FILE}
    
    echo "    数据库schema导入完成"
}

# 函数: inject_bigtop_stack
# 功能: 注入Bigtop 3.5.0 Stack定义
inject_bigtop_stack() {
    echo "==> [步骤6] 注入Bigtop 3.5.0 Stack..."
    
    # 查找父Stack版本
    # 列出BIGTOP目录下的所有版本，排除3.5.0，取最新的
    PARENT_STACK=$(ls "${STACKS_DIR}/BIGTOP/" | grep -v '3.5.0' | sort -V | tail -1)
    echo "    父Stack版本: BIGTOP/${PARENT_STACK}"
    
    # 复制父Stack作为基础
    cp -r "${STACKS_DIR}/BIGTOP/${PARENT_STACK}" "${STACK_HOME}"
    
    # 创建metainfo.xml
    # 指定Stack继承关系
    cat > "${STACK_HOME}/metainfo.xml" << EOF
<?xml version="1.0"?>
<metainfo>
  <versions>
    <active>true</active>
  </versions>
  <extends>${PARENT_STACK}</extends>
</metainfo>
EOF
    
    # 创建仓库配置
    mkdir -p "${STACK_HOME}/repos"
    cat > "${STACK_HOME}/repos/repoinfo.xml" << 'EOF'
<?xml version="1.0"?>
<reposinfo>
  <latest>http://bigtop_hostname0/repodata/repomd.xml</latest>
  <os family="redhat8">
    <repo>
      <baseurl>http://bigtop_hostname0</baseurl>
      <repoid>BIGTOP-3.5.0</repoid>
      <reponame>Bigtop 3.5.0</reponame>
      <unique>true</unique>
    </repo>
  </os>
</reposinfo>
EOF
    
    echo "    Stack注入完成"
}

# 函数: update_service_versions

    echo "==> [步骤7] 更新服务版本号..."
```

```bash
#!/bin/bash
#===============================================================================
# 文件名称: fix-hive.sh

# 版本: V1.0
#
# 问题说明:
#   1. hive.exec.post.hooks配置值为空格而非空字符串
#   2. service_advisor.py返回空格而非空字符串
#   3. hive_check.py使用auth=noSasl而非auth=none
#   4. HDFS目录权限不正确
#   5. Metastore schema未初始化
#   6. 缺少sys.db.created标记文件
#
# 使用方法:
#   ./fix-hive.sh
#
# 前置条件:
#   1. Hive服务已部署
#   2. HDFS服务已启动
#
# 执行结果:
#   所有兼容性问题修复完成

# 功能: 修复hive-site.xml中的配置问题
fix_hive_site() {
    echo "[1/7] 修复hive-site.xml配置..."
    
    if [ -f "$HIVE_SITE" ]; then
        # 备份原文件
        cp "$HIVE_SITE" "${HIVE_SITE}.bak"
        
        # 修复hive.exec.post.hooks空格值问题
        # 将 <value> </value> 替换为 <value></value>
        sed -i 's|<value> </value>|<value></value>|g' "$HIVE_SITE"
        
        echo "    已修复hive.exec.post.hooks配置"
    else
        echo "    警告: hive-site.xml未找到"
    fi
}

# 函数: fix_service_advisor
# 功能: 修复service_advisor.py中的bug
fix_service_advisor() {
    echo "[2/7] 修复service_advisor.py..."
    
    # 查找HIVE service_advisor.py文件
    local ADVISOR=$(find ${STACKS_DIR} -name "service_advisor.py" -path "*/HIVE/*" | head -1)
    
    if [ -n "$ADVISOR" ] && [ -f "$ADVISOR" ]; then
        # 备份原文件
        cp "$ADVISOR" "${ADVISOR}.bak"
        
        # 修复空格值bug
        # 当没有hooks时，返回空字符串而非空格
        sed -i 's/return " "/return ""/g' "$ADVISOR"
        
        echo "    已修复: $ADVISOR"
    else
        echo "    警告: service_advisor.py未找到"
    fi
}

# 函数: fix_hive_check
# 功能: 修复hive_check.py中的认证参数问题
fix_hive_check() {
    echo "[3/7] 修复hive_check.py..."
    
    local HIVE_CHECK="${AGENT_LIB}/hive_check.py"
    
    if [ -f "$HIVE_CHECK" ]; then
        # 备份原文件
        cp "$HIVE_CHECK" "${HIVE_CHECK}.bak"
        
        # 修复认证参数
        # Hive 4.0.1要求使用auth=none而非auth=noSasl
        sed -i 's/beeline_url.append("auth=noSasl")/beeline_url.append("auth=none")/g' "$HIVE_CHECK"
        
        echo "    已修复认证参数: auth=noSasl -> auth=none"
    else
        echo "    警告: hive_check.py未找到"
    fi
}

# 函数: fix_hdfs_permissions
# 功能: 修复HDFS目录权限
fix_hdfs_permissions() {
    echo "[4/7] 修复HDFS目录权限..."
    
    # 检查hdfs命令是否可用
    if ! command -v hdfs &> /dev/null; then
        echo "    警告: hdfs命令不可用，跳过HDFS权限修复"
        return
    fi
    
    # 创建Hive仓库目录
    # managed: 托管表目录
    # external: 外部表目录
    hdfs dfs -mkdir -p /warehouse/tablespace/managed/hive 2>/dev/null || true
    hdfs dfs -mkdir -p /warehouse/tablespace/external/hive 2>/dev/null || true
    hdfs dfs -mkdir -p /tmp/hive 2>/dev/null || true
    
    # 设置目录所有者和权限
    # hive:hadoop: hive用户，hadoop组
    hdfs dfs -chown -R hive:hadoop /warehouse/tablespace 2>/dev/null || true
    hdfs dfs -chmod -R 775 /warehouse/tablespace 2>/dev/null || true
    hdfs dfs -chown hive:hadoop /tmp/hive 2>/dev/null || true
    hdfs dfs -chmod 777 /tmp/hive 2>/dev/null || true
    
    echo "    HDFS目录权限已修复"
}

# 函数: init_metastore_schema
# 功能: 初始化Hive Metastore schema

    echo "[5/7] 检查Metastore schema..."

        echo "    警告: mysql命令不可用，跳过schema初始化"

        echo "    Metastore schema已存在: $SCHEMA_VERSION"

        echo "    Metastore schema已初始化"

    fi
```

### 3.1 insert-repo-version.py - 版本信息注入工具

```python
#!/usr/bin/env python3
"""
文件名称: insert-repo-version.py
功能描述: HodoopForge版本信息数据库注入工具
         用于向Ambari数据库注入Bigtop 3.5.0版本仓库信息

作者: HodoopForge开发团队
创建日期: 2026-04
版本: V1.0

使用方法:
    python3 insert-repo-version.py

前置条件:
    1. MySQL数据库已安装
    2. Ambari数据库已创建
    3. VDF文件已生成

执行结果:
    在repo_version表中插入Bigtop 3.5.0版本记录
"""

# 导入必要的模块
import mysql.connector  # MySQL数据库连接器
import sys              # 系统相关功能
import os               # 操作系统接口

#===============================================================================
# 配置常量
#===============================================================================

# VDF文件路径
# VDF (Version Definition File) 定义了版本信息和仓库地址
VDF_PATH = '/var/repo/ambari/bigtop-3.5.0-vdf.xml'

# 数据库连接配置
DB_CONFIG = {
    'host': 'localhost',      # 数据库主机
    'port': 3306,             # 数据库端口
    'user': 'ambari',         # 数据库用户名
    'password': 'ambari',     # 数据库密码
    'database': 'ambari'      # 数据库名称
}

# 版本信息
VERSION_INFO = {
    'repo_version_id': 1,                    # 版本ID
    'version': '3.5.0.0-1',                  # 版本号
    'display_name': 'Bigtop-3.5.0.0-1',      # 显示名称
    'repo_type': 'STANDARD',                 # 仓库类型
    'hidden': 0,                             # 是否隐藏
    'resolved': 1,                           # 是否已解析
    'legacy': 0,                             # 是否旧版本
    'version_url': 'http://bigtop_hostname0/bigtop-3.5.0-vdf.xml'  # VDF URL
}

#===============================================================================
# 主函数
#===============================================================================

def main():
    """
    主函数：执行版本信息注入
    
    执行流程:
        1. 检查VDF文件是否存在
        2. 读取VDF文件内容
        3. 连接数据库
        4. 检查版本是否已存在
        5. 插入版本记录
    """
    
    print("==> HodoopForge: 注入版本信息...")
    
    #---------------------------------------------------------------------------
    # 步骤1: 检查VDF文件
    #---------------------------------------------------------------------------
    if not os.path.exists(VDF_PATH):
        print(f"错误: VDF文件不存在: {VDF_PATH}")
        print("请先执行仓库构建脚本生成VDF文件")
        sys.exit(1)
    
    #---------------------------------------------------------------------------
    # 步骤2: 读取VDF文件内容
    #---------------------------------------------------------------------------
    print("    读取VDF文件...")
    with open(VDF_PATH, 'r', encoding='utf-8') as f:
        vdf_xml = f.read()
    
    print(f"    VDF文件大小: {len(vdf_xml)} 字节")
    
    #---------------------------------------------------------------------------
    # 步骤3: 连接数据库
    #---------------------------------------------------------------------------
    print("    连接数据库...")
    
    try:
        # 创建数据库连接
        conn = mysql.connector.connect(**DB_CONFIG)
        cursor = conn.cursor()
        
        print("    数据库连接成功")
        
    except mysql.connector.Error as err:
        print(f"错误: 数据库连接失败: {err}")
        sys.exit(1)
    
    #---------------------------------------------------------------------------
    # 步骤4: 检查版本是否已存在
    #---------------------------------------------------------------------------
    print("    检查版本是否已存在...")
    
    try:
        # 查询repo_version表
        cursor.execute(
            "SELECT COUNT(*) FROM repo_version WHERE version = %s",
            (VERSION_INFO['version'],)
        )
        
        count = cursor.fetchone()[0]
        
        if count > 0:
            print(f"    版本 {VERSION_INFO['version']} 已存在，跳过插入")
            cursor.close()
            conn.close()
            return
        
    except mysql.connector.Error as err:
        print(f"错误: 查询失败: {err}")
        cursor.close()
        conn.close()
        sys.exit(1)
    
    #---------------------------------------------------------------------------
    # 步骤5: 获取stack_id
    #---------------------------------------------------------------------------
    print("    获取stack_id...")
    
    try:
        # 查询BIGTOP stack的ID
        cursor.execute(
            "SELECT stack_id FROM stack WHERE stack_name = 'BIGTOP'"
        )
        
        result = cursor.fetchone()
        
        if result:
            stack_id = result[0]
        else:
            # 如果找不到，使用默认值
            stack_id = 4
            print(f"    警告: 未找到BIGTOP stack，使用默认ID: {stack_id}")
        
        print(f"    stack_id = {stack_id}")
        
    except mysql.connector.Error as err:
        print(f"警告: 查询stack_id失败: {err}")
        stack_id = 4
    
    #---------------------------------------------------------------------------
    # 步骤6: 插入版本记录
    #---------------------------------------------------------------------------
    print("    插入版本记录...")
    
    try:
        # 构建INSERT语句
        insert_sql = """
            INSERT INTO repo_version
            (repo_version_id, stack_id, version, display_name, repo_type,
             hidden, resolved, legacy, version_url, version_xml)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        """
        
        # 执行插入
        cursor.execute(insert_sql, (
            VERSION_INFO['repo_version_id'],
            stack_id,
            VERSION_INFO['version'],
            VERSION_INFO['display_name'],
            VERSION_INFO['repo_type'],
            VERSION_INFO['hidden'],
            VERSION_INFO['resolved'],
            VERSION_INFO['legacy'],
            VERSION_INFO['version_url'],
            vdf_xml
        ))
        
        # 提交事务
        conn.commit()
        
        print(f"    版本 {VERSION_INFO['version']} 插入成功")
        
    except mysql.connector.Error as err:
        print(f"错误: 插入失败: {err}")
        conn.rollback()
        cursor.close()
        conn.close()
        sys.exit(1)
    
    #---------------------------------------------------------------------------
    # 步骤7: 清理资源
    #---------------------------------------------------------------------------
    cursor.close()
    conn.close()
    
    print("==> 版本信息注入完成")


#===============================================================================
# 程序入口
#===============================================================================

if __name__ == '__main__':
    main()
```
