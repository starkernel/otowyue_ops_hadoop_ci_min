### 2.1 setup.sh - 节点初始化脚本（核心部分）

```bash
#!/bin/bash
#
# HodoopForge - 专为现代Hadoop集群量身打造的一站式部署与管理平台
# Forge Stability. Connect the Future.
#
# This script is part of the HodoopForge project.
# Project: HodoopForge - Ambari 3.x + Bigtop 3.5.0 Integration Platform
#

set -ex

INSTANCES="bigtop_hostname0 bigtop_hostname1 bigtop_hostname2 bigtop_hostname3"

for i in $INSTANCES ; do
  if [ "$(docker compose ps | grep -c $i)" == "0" ] ; then
    echo "can't find docker container: $i"
    exit 1
  fi
done

for i in $INSTANCES ; do
  docker compose exec -it $i dnf install -y sudo openssh-server openssh-clients which iproute net-tools less vim-enhanced
  docker compose exec -it $i dnf install -y initscripts wget curl tar unzip git
  docker compose exec -it $i dnf install -y dnf-plugins-core
  docker compose exec -it $i dnf config-manager --set-enabled powertools 2>/dev/null \
    || docker compose exec -it $i dnf config-manager --set-enabled rocky-powertools 2>/dev/null \
    || echo "powertools already enabled or not needed"
  docker compose exec -it $i dnf update -y || echo "update failed, but continuing"
done

# SSH密钥配置
for i in $INSTANCES ; do
  docker compose exec -it $i bash -c \
    '[ -f /root/.ssh/id_rsa ] || ssh-keygen -t rsa -N "" -f /root/.ssh/id_rsa'
  docker compose exec -it $i systemctl enable sshd
  docker compose exec -it $i systemctl start sshd
done

# 分发公钥到所有节点
docker compose cp bigtop_hostname0:/root/.ssh/id_rsa.pub bigtop_hostname0.pub

for i in $INSTANCES ; do
  docker compose cp bigtop_hostname0.pub $i:/root/.ssh/authorized_keys
  docker compose exec -it $i chown root:root /root/.ssh/authorized_keys
  docker compose exec -it $i chmod 600 /root/.ssh/authorized_keys
done
```

### 2.2 setup-bigtop-stack.sh - Stack注入脚本（核心部分）

```bash
#!/bin/bash
#
# HodoopForge - 专为现代Hadoop集群量身打造的一站式部署与管理平台
# Forge Stability. Connect the Future.
#

set -ex

STACKS_DIR=/var/lib/ambari-server/resources/stacks
STACK_HOME=${STACKS_DIR}/BIGTOP/3.5.0

# 动态继承现有Stack
PARENT_STACK=$(ls "${STACKS_DIR}/BIGTOP/" | grep -v '3.5.0' | sort -V | tail -1)

echo "==> Found parent stack BIGTOP/${PARENT_STACK}, creating BIGTOP/3.5.0..."

# 复制父Stack作为基础
cp -r "${STACKS_DIR}/BIGTOP/${PARENT_STACK}" "${STACK_HOME}"

# 更新metainfo.xml
cat > "${STACK_HOME}/metainfo.xml" << 'EOF'
<?xml version="1.0"?>
<metainfo>
  <versions>
    <active>true</active>
  </versions>
  <extends>PARENT_VERSION</extends>
</metainfo>
EOF

sed -i "s/PARENT_VERSION/${PARENT_STACK}/" "${STACK_HOME}/metainfo.xml"

# 配置仓库信息
mkdir -p "${STACK_HOME}/repos"
cat > "${STACK_HOME}/repos/repoinfo.xml" << 'EOF'
<?xml version="1.0"?>
<reposinfo>
  <latest>http://bigtop_hostname0/repodata/repomd.xml</latest>
  <os family="redhat8">
    <repo>
      <baseurl>http://bigtop_hostname0</baseurl>
      <repoid>BIGTOP-3.5.0</repoid>
      <reponame>Bigtop 3.5.0</reponame>
      <unique>true</unique>
    </repo>
  </os>
</reposinfo>
EOF

# 更新组件版本
update_service_version() {
  local service=$1
  local pkg=$2
  local version=$3
  local svc_dir="${STACK_HOME}/services/${service}"

  [ -d "${svc_dir}" ] || return 0

  sed -i "s|<version>.*</version>|<version>${version}</version>|g" \
    "${svc_dir}/metainfo.xml" 2>/dev/null || true
}
```

```bash
#!/bin/bash
#
# HodoopForge - 专为现代Hadoop集群量身打造的一站式部署与管理平台
# Forge Stability. Connect the Future.
#

set -e

echo "==> HodoopForge Hive Service Fix Script"

# 修复 hive.exec.post.hooks 配置
echo "[1/6] Fixing hive.exec.post.hooks configuration..."
if [ -f /etc/hive/conf/hive-site.xml ]; then
  sed -i 's|<value> </value>|<value></value>|g' /etc/hive/conf/hive-site.xml
  echo "    Fixed hive.exec.post.hooks empty value"
fi

# 修复 Ambari Agent hive_check.py auth 参数
echo "[3/7] Fixing Ambari Agent hive_check.py auth parameter..."
HIVE_CHECK="/usr/lib/ambari-agent/lib/resource_management/libraries/functions/hive_check.py"
if [ -f "$HIVE_CHECK" ]; then
  sed -i 's/beeline_url.append("auth=noSasl")/beeline_url.append("auth=none")/g' "$HIVE_CHECK"
  echo "    Fixed hive_check.py auth parameter"
fi

# 修复 HDFS 目录权限
echo "[4/7] Fixing HDFS directory permissions for Hive..."
hdfs dfs -mkdir -p /warehouse/tablespace/managed/hive 2>/dev/null || true
hdfs dfs -mkdir -p /warehouse/tablespace/external/hive 2>/dev/null || true
hdfs dfs -chown -R hive:hadoop /warehouse/tablespace 2>/dev/null || true

# 初始化 Metastore schema
echo "[5/7] Checking Hive Metastore schema..."
SCHEMA_VERSION=$(mysql -uhive -phive hive -N -e "SELECT SCHEMA_VERSION FROM VERSION" 2>/dev/null || echo "")
if [ -z "$SCHEMA_VERSION" ]; then
  if [ -f /usr/lib/hive/scripts/metastore/upgrade/mysql/hive-schema-4.0.0.mysql.sql ]; then
    mysql -uhive -phive hive < /usr/lib/hive/scripts/metastore/upgrade/mysql/hive-schema-4.0.0.mysql.sql 2>/dev/null || true
    echo "    Schema initialized"
  fi
fi
```

### 4.2 api-client.py - Ambari REST API客户端

```python
#!/usr/bin/env python3
# ============================================================
# 软件名称: HodoopForge 企业级Hadoop集群部署与管理平台软件
# 文件名称: api-client.py
# 文件功能: Ambari REST API 客户端
# 版    本: V1.0
# 编制日期: 2026年4月
# ============================================================

import requests
import json
from typing import Dict, List, Optional

class AmbariAPIClient:
    """Ambari REST API 客户端类"""
    
    def __init__(self, host: str = "localhost", port: int = 20609,
                 username: str = "admin", password: str = "admin"):
        """
        初始化 API 客户端
        
        参数:
            host: Ambari Server 主机名
            port: Ambari Server 端口
            username: 认证用户名
            password: 认证密码
        """
        self.base_url = f"http://{host}:{port}/api/v1"
        self.auth = (username, password)
        self.session = requests.Session()
        self.session.auth = self.auth
    
    def get_clusters(self) -> List[Dict]:
        """获取所有集群列表"""
        return self.get("clusters").get("items", [])
    
    def get_hosts(self, cluster_name: str) -> List[Dict]:
        """获取集群所有节点"""
        return self.get(f"clusters/{cluster_name}/hosts").get("items", [])
    
    def start_service(self, cluster_name: str, service_name: str) -> Dict:
        """启动指定服务"""
        data = {
            "RequestInfo": {"context": "启动服务"},
            "Body": {"ServiceInfo": {"state": "STARTED"}}
        }
        return self.put(f"clusters/{cluster_name}/services/{service_name}", data)
```

```python
#!/usr/bin/env python3
# ============================================================
# 软件名称: HodoopForge 企业级Hadoop集群部署与管理平台软件
# 文件名称: cluster-scaler.py
# 文件功能: 集群扩缩容工具
# 版    本: V1.0
# 编制日期: 2026年4月
# ============================================================

import requests
import json
from typing import List, Dict

class ClusterScaler:
    """集群扩缩容管理器"""
    
    def __init__(self, ambari_host: str = "localhost",
                 ambari_port: int = 20609,
                 username: str = "admin", password: str = "admin"):
        """初始化扩缩容管理器"""
        self.base_url = f"http://{ambari_host}:{ambari_port}/api/v1"
        self.auth = (username, password)
    
    def add_host(self, cluster_name: str, host_name: str) -> bool:
        """添加节点到集群"""
        url = f"{self.base_url}/clusters/{cluster_name}/hosts/{host_name}"
        data = {"Hosts": {"host_name": host_name}}
        response = requests.post(url, json=data, auth=self.auth)
        return response.status_code == 201
    
    def remove_host(self, cluster_name: str, host_name: str) -> bool:
        """从集群移除节点"""
        url = f"{self.base_url}/clusters/{cluster_name}/hosts/{host_name}"
        response = requests.delete(url, auth=self.auth)
        return response.status_code == 200
    
    def rebalance_hdfs(self, cluster_name: str) -> bool:
        """重新平衡 HDFS 数据"""
        url = f"{self.base_url}/clusters/{cluster_name}/requests"
        data = {
            "RequestInfo": {
                "context": "HDFS 重新平衡",
                "command": "REBALANCE_HDFS"
            }
        }
        response = requests.post(url, json=data, auth=self.auth)
        return response.status_code == 202
```

### 4.4 backup-restore.py - 备份恢复工具

```python
#!/usr/bin/env python3
# ============================================================
# 软件名称: HodoopForge 企业级Hadoop集群部署与管理平台软件
# 文件名称: backup-restore.py
# 文件功能: 备份恢复工具
# 版    本: V1.0
# 编制日期: 2026年4月
# ============================================================

import os
import tarfile
import subprocess
from datetime import datetime
from typing import Optional

class BackupRestore:
    """备份恢复管理器"""
    
    def __init__(self, backup_dir: str = "/backup"):
        """初始化备份恢复管理器"""
        self.backup_dir = backup_dir
        os.makedirs(backup_dir, exist_ok=True)
    
    def backup_config(self, config_paths: list, 
                     backup_name: Optional[str] = None) -> str:
        """备份配置文件"""
        if backup_name is None:
            backup_name = f"config-{datetime.now().strftime('%Y%m%d%H%M%S')}"
        
        backup_file = os.path.join(self.backup_dir, f"{backup_name}.tar.gz")
        
        with tarfile.open(backup_file, "w:gz") as tar:
            for path in config_paths:
                if os.path.exists(path):
                    tar.add(path, arcname=os.path.basename(path))
        
        return backup_file
    
    def backup_database(self, db_name: str, db_user: str, 
                       db_pass: str, backup_name: Optional[str] = None) -> str:
        """备份数据库"""
        if backup_name is None:
            backup_name = f"db-{db_name}-{datetime.now().strftime('%Y%m%d%H%M%S')}"
        
        backup_file = os.path.join(self.backup_dir, f"{backup_name}.sql")
        
        cmd = f"mysqldump -u{db_user} -p{db_pass} {db_name} > {backup_file}"
        subprocess.run(cmd, shell=True, check=True)
        
        return backup_file
```
